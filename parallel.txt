source .venv/bin/activate
export VENV_PY="$(which python)"
export BASE="/media/nvme7n1/fbtelki/dyldo_full/data"
mkdir -p summaries logs

find "$BASE" -maxdepth 1 -type d -regextype posix-extended -regex '.*/[0-9]{4}-[0-9]{2}-[0-9]{2}$' \
| sort \
| xargs -I{} -P 12 bash -lc '
  W=$(basename "{}")
  IN1="{}/data.nq.gz"
  IN2="{}/data-seedlist.nq.gz"
  OUT="summaries/${W}_attributeclass_onlyid.json"
  LOG="logs/${W}.log"

  # Skip if already good
  if [[ -s "$OUT" ]] && jq -e ".nodes|length>0 and .edges|length>=0" "$OUT" >/dev/null 2>&1; then
    echo "[$W] already done"
    exit 0
  fi

  # Note missing/corrupt files, but still run if at least one survives
  [[ -f "$IN1" ]] || echo "[$W] missing $IN1" | tee -a logs/_missing.log
  [[ -f "$IN2" ]] || echo "[$W] missing $IN2" | tee -a logs/_missing.log
  if [[ -f "$IN1" ]] && ! gzip -t "$IN1" >/dev/null 2>&1; then echo "[$W] CORRUPT: $IN1" | tee -a logs/_corrupt.log; IN1=""; fi
  if [[ -f "$IN2" ]] && ! gzip -t "$IN2" >/dev/null 2>&1; then echo "[$W] CORRUPT: $IN2" | tee -a logs/_corrupt.log; IN2=""; fi
  [[ -n "$IN1" || -n "$IN2" ]] || { echo "[$W] all inputs missing/corrupt → skip"; exit 0; }

  # Build argv safely (NO embedded quotes)
  args=( --input )
  [[ -n "$IN1" ]] && args+=( "$IN1" )
  [[ -n "$IN2" ]] && args+=( "$IN2" )

  echo "[$W] starting" | tee "$LOG"
  "'"$VENV_PY"'" main_build.py \
      "${args[@]}" \
      --strategy attributeclass \
      --out summaries/ \
      --only-id --use-numeric-ids --limit 0 \
      >>"$LOG" 2>&1

  rc=$?
  if (( rc == 0 )); then
    # sanity: flag empty payloads so we can redo later
    if jq -e ".nodes|length==0 and .edges|length==0" "$OUT" >/dev/null 2>&1; then
      echo "[$W] ⚠ empty payload (0 nodes/edges) — will redo" | tee -a logs/_empty_payload.log
    else
      echo "[$W] ✔ done"
    fi
  else
    echo "[$W] ✖ failed (rc=$rc) — see $LOG"
  fi
'
